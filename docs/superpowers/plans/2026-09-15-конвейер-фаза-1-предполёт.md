# Конвейер карусели, фаза 1: предполёт — план

> **Для исполнителя:** обязательный скил — `subagent-driven-development` (рекомендуется) или `executing-plans`. Шаги отмечаются чекбоксами (`- [ ]`).

**Цель:** сломанные связки вёрсток и текстовые ловушки ловятся и чинятся до первого кадра, а не на готовой карусели.

**Устройство:** в `форматы.py` появляется матрица запретов со ссылками на реестр дефектов `docs/дефекты-кадров.md`. Паспорт не выбирает дизайн, негодный формату, а задание модели называет вёрстки, которые в этом дизайне ломаются. Новый `предполёт.py` без модели и Chrome проверяет текст карусели и чинит беды: в эскизе разрешён один проход модели (`carousel.переверстать`), в дорисовке только правка без модели. Отчёт ложится в карточку и показывается придирками под черновиком. Скил `конвейер-карусели` ведёт сессию Claude по этапам.

**Стек:** Python 3, unittest, pydantic, существующие `card.py`, `carousel.py`, `core.py`, `придирки.py`.

**Спека:** `docs/superpowers/specs/2026-09-15-конвейер-карусели-design.md`.

```python
# -*- coding: utf-8 -*-
"""Матрица связок: какие вёрстки в каком дизайне ломаются, и реестр под ними.

Сломанная связка видна ещё по тексту: «было» в каталоге на тёмном кадре
выходит белым по крему всегда. Матрица говорит об этом до отрисовки, а реестр
хранит, откуда запрет взялся, чтобы его сняли вместе с починкой, а не забыли.
"""
import re
import sys
import unittest
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

import форматы  # noqa: E402

РЕЕСТР = Path(__file__).resolve().parent.parent / "docs" / "дефекты-кадров.md"
СТАТУСЫ = ("открыт", "обход в матрице", "обход в предполёте",
           "починен с тестом", "автопроверка")


class Матрица(unittest.TestCase):
    def test_каталог_на_тёмном_кадре_не_берёт_было(self):
        self.assertEqual(форматы.запрет("каталог", "было", 2, "3к4"), "Д-01")
        self.assertEqual(форматы.запрет("каталог", "стрелка", 4, "квадрат"), "Д-01")

    def test_каталог_на_светлом_кадре_было_можно(self):
        self.assertEqual(форматы.запрет("каталог", "было", 3, "3к4"), "")

    def test_против_в_каталоге_нельзя_нигде(self):
        for номер in (2, 3):
            with self.subTest(номер=номер):
                self.assertEqual(форматы.запрет("каталог", "против", номер, "3к4"), "Д-02")

    def test_шаг_в_пособии_нельзя(self):
        self.assertEqual(форматы.запрет("пособие", "шаг", 5, "3к4"), "Д-04")

    def test_манифест_в_квадрате_нельзя_целиком(self):
        self.assertEqual(форматы.запрет("манифест", "мысль", 3, "квадрат"), "Д-05")
        self.assertFalse(форматы.дизайн_годится("манифест", "квадрат"))
        self.assertTrue(форматы.дизайн_годится("манифест", "3к4"))
        self.assertTrue(форматы.дизайн_годится("каталог", "квадрат"))

    def test_фирменный_без_запретов(self):
        self.assertEqual(форматы.запрет("фирменный", "против", 2, "квадрат"), "")

    def test_надёжный_вид_не_запрещён_ни_в_одной_связке(self):
        for п in форматы.НЕЛЬЗЯ:
            with self.subTest(дефект=п["дефект"]):
                if "виды" in п:
                    self.assertNotIn(форматы.НАДЁЖНЫЙ_ВИД, п["виды"])
                else:
                    self.assertTrue(п.get("формат") and not п.get("тон"),
                                    "дизайн целиком запрещается только по формату")

    def test_статус_проверено_берётся_из_образцов(self):
        было = set(форматы.ПРОВЕРЕНО)
        try:
            форматы.ПРОВЕРЕНО.add(("терминал", "список", "тёмный", "3к4"))
            self.assertEqual(форматы.статус("терминал", "список", 2, "3к4"), "проверено")
            self.assertEqual(форматы.статус("терминал", "список", 3, "3к4"), "не проверено")
            self.assertEqual(форматы.статус("каталог", "против", 3, "3к4"), "нельзя")
        finally:
            форматы.ПРОВЕРЕНО.clear()
            форматы.ПРОВЕРЕНО.update(было)


class Реестр(unittest.TestCase):
    def setUp(self):
        self.текст = РЕЕСТР.read_text(encoding="utf-8")
        self.записи = set(re.findall(r"^## (Д-\d\d)\.", self.текст, re.M))

    def test_каждый_запрет_записан_в_реестре(self):
        дефекты = {п["дефект"] for п in форматы.НЕЛЬЗЯ}
        дефекты |= {дефект for _, дефект in форматы.БЕЗ_ПОДСВЕТКИ.values()}
        self.assertEqual(дефекты - self.записи, set())

    def test_у_каждой_записи_есть_статус(self):
        for кусок in re.split(r"^## ", self.текст, flags=re.M)[1:]:
            with self.subTest(запись=кусок.splitlines()[0]):
                статус = re.search(r"^- Статус: (.+)$", кусок, re.M)
                self.assertIsNotNone(статус)
                self.assertIn(статус.group(1).strip(), СТАТУСЫ)


if __name__ == "__main__":
    unittest.main()
```

```python
# строка отсюда уходит, иначе запрет переживёт причину и молча сузит выбор.
# «тон» — тёмный или светлый кадр по card.светлый_ли, без него — любой кадр.
# Без «видов» запрещён дизайн целиком, и только по формату.
НЕЛЬЗЯ = [
    {"дефект": "Д-01", "дизайн": "каталог", "тон": "тёмный",
     "виды": {"было", "стрелка"}},
    {"дефект": "Д-02", "дизайн": "каталог", "виды": {"против"}},
    {"дефект": "Д-04", "дизайн": "пособие", "виды": {"шаг"}},
    {"дефект": "Д-05", "дизайн": "манифест", "формат": "квадрат"},
]

# Подсветка *звёздочками*, которая в дизайне пропадает на кадрах одного тона.
# Вёрстку тут менять незачем: слова видны, не видна только заливка.
БЕЗ_ПОДСВЕТКИ = {"пособие": ("светлый", "Д-03")}

# Куда переводится слайд из сломанной связки. В НЕЛЬЗЯ стоять не может: тест
# сторожит, иначе переводить станет некуда.
НАДЁЖНЫЙ_ВИД = "мысль"

# Связки, проверенные на образцах: (дизайн, вид, тон, формат). Наполняет
# прогон образцов; до него «не проверено» рисовать разрешено.
ПРОВЕРЕНО: set = set()


def тон(номер: int) -> str:
    import card
    return "светлый" if card.светлый_ли(номер) else "тёмный"


def запрет(дизайн: str, вид: str, номер: int, формат: str) -> str:
    """Номер дефекта, если связка ломается; пустая строка, если рисовать можно."""
    for п in НЕЛЬЗЯ:
        if п["дизайн"] != дизайн:
            continue
        if п.get("формат") and п["формат"] != формат:
            continue
        if п.get("тон") and п["тон"] != тон(номер):
            continue
        if "виды" not in п or вид in п["виды"]:
            return п["дефект"]
    return ""


def статус(дизайн: str, вид: str, номер: int, формат: str) -> str:
    if запрет(дизайн, вид, номер, формат):
        return "нельзя"
    if (дизайн, вид, тон(номер), формат) in ПРОВЕРЕНО:
        return "проверено"
    return "не проверено"


def дизайн_годится(дизайн: str, формат: str) -> bool:
    """Дизайн целиком не годится формату — как манифест в квадрате."""
    return not any(п["дизайн"] == дизайн and "виды" not in п
                   and not п.get("тон") and п.get("формат") == формат
                   for п in НЕЛЬЗЯ)
```

```python
ВИДЫ_СПИСКА = {"список", "шпаргалка", "поток"}
```

```python
СОЕДИНИТЬ = {"было": " → ", "стрелка": " → ", "против": " или "}


def _в_надёжный(с: dict) -> dict:
    """Слайд из сломанной связки в надёжный вид, не теряя слов.

    Пункты переезжают в текст, от «что|сколько|доля» остаётся «что»: числа
    полосой без полосы не читаются. Номер шага уходит: он дублирует порядок
    слайдов, а в мысли выглядел бы опечаткой.
    """
    вид = с.get("вид") or ""
    пункты = [str(х).split("|")[0].strip() for х in (с.get("строки") or [])
              if str(х).strip()]
    куски = [str(с.get("текст") or "").strip(), СОЕДИНИТЬ.get(вид, "; ").join(пункты)]
    заголовок = str(с.get("заголовок") or "").strip() or str(с.get("значение") or "").strip()
    return {**с, "вид": форматы.НАДЁЖНЫЙ_ВИД, "заголовок": заголовок,
            "текст": " ".join(кусок for кусок in куски if кусок),
            "значение": "", "строки": []}


def исправить(к: dict, формат: str) -> tuple[dict, list[str]]:
    """Чинит беды предполёта без модели. Новая карусель и что сделано.

    Слова не выбрасываются: сломанная вёрстка переводится в надёжную, пункты
    переезжают в текст. Текст выходит длиннее задуманного — это не брак:
    рисовальщик ужмёт кегль, полная генерация сократит. Дизайн, негодный
    формату, вёрсткой не чинится — это дело паспорта, и слайды не трогаются.
    """
    import carousel
    паспорт = к.get("паспорт") or {}
    дизайн = паспорт.get("дизайн") or "фирменный"
    if not форматы.дизайн_годится(дизайн, формат):
        return к, []
    начало = 1 if carousel.одним_кадром(паспорт) else 2
    слайды, сделано = [], []
    for номер, с in enumerate(к.get("slides") or [], start=начало):
        с = dict(с)
        вид = с.get("вид") or форматы.НАДЁЖНЫЙ_ВИД
        if форматы.запрет(дизайн, вид, номер, формат):
            с = _в_надёжный(с)
            сделано.append(f"№{номер}: «{вид}» → «{форматы.НАДЁЖНЫЙ_ВИД}»")
        без = форматы.БЕЗ_ПОДСВЕТКИ.get(дизайн)
        if (без and без[0] == форматы.тон(номер)
                and ЗВЁЗДОЧКИ.search(str(с.get("текст") or ""))):
            с["текст"] = ЗВЁЗДОЧКИ.sub(r"\1", str(с["текст"]))
            сделано.append(f"№{номер}: снята подсветка")
        слайды.append(с)
    return {**к, "slides": слайды}, сделано
```

```python
def довести(к: dict, формат: str, модель: bool = False) -> tuple[dict, dict]:
    """Проверяет и чинит до отрисовки. Модель — не больше одного прохода.

    Модель разрешают только для нового текста, который ещё никто не видел.
```

```markdown
проходит предполёт (сломанную вёрстку модель перекладывает один раз, остальное
чинится без неё):

    nohup $П -c 'import core; print(core.эскиз(None, None, "tiktok", "<указания>"))' > /tmp/эскиз.log 2>&1 &

Показать владельцу текстом, по слайдам с видом каждого, и строку предполёта:

    $П -c 'import db, предполёт; п=db.карточка(db.взять(ИД)); print(предполёт.строкой(п.get("предполёт")))'

Экрана эскиза в боте пока нет (фаза 3 спеки): сценарии показываются в чате.

**3. Отрисовка.** Только после «рисовать», по одной, фоном:

    nohup $П -c 'import core; core.дорисовать(ИД)' > /tmp/дорисовка.log 2>&1 &

**4. Приёмка автоматикой.** Сперва текст, картинки потом:

    $П tools_лист.py ИД

Выводит путь листа кадров и «годен» или «брак: №N: …».

**5. Осмотр по чек-листу — одна картинка листа, не десять кадров.** Отметить
«годен» может владелец или Claude.
1. Нет пустых и полупустых кадров.
2. На светлых кадрах видны подсветка и цифры.
3. Кот не срезан рамкой, не накрывает текст, лапы и хвост в норме (Д-10).
4. Хвост буквы финала не заходит на плашку призыва — это брак (Д-07).
5. Тёмные и светлые кадры чередуются, соседние свёрстаны по-разному.
6. Призыв и финал подходят именно к этой карусели (Д-08).
7. Обещание обложки выполнено внутри.

**6. Правка кадра по лестнице.** Правится кадр, остальные не трогаются.
- Брак окружения (`страница: …`) — та же отрисовка ещё раз, это не круг.
- Круг 1: текст слайда в `card["carousel"]["slides"][N-2]`, `db.правка(ид, card=...)`,
  затем `$П -c 'import jobs; jobs.поставить("перерисовать", {"id": ИД, "слайд": N})'`.
  Поза — то же задание с `"брак": True`.
- Круг 2: слайд переводится на надёжную вёрстку («мысль» или «список»).
- После двух кругов карусель **не выбрасывается**: кадр убирается или сливается
  с соседним, если длина и обещание обложки сходятся; иначе черновик ждёт починки.
  Решает владелец или Claude, случай записывается в реестр дефектов.
- Баг рисовальщика текстом обходится один раз, дальше — запись в реестр и Контур Р.

**7. Выпуск.** Годен по приёмке и осмотру — папка на столе собирается сама.
После выкладки владелец жмёт «Выложил руками».
```
