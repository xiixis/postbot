# Новый интерфейс приложения — план реализации

> **Для исполнителя:** шаги отмечаются галочками `- [ ]`. Порядок задач важен:
> сервер раньше клиента, каркас раньше экранов.

**Цель.** Дать приложению прошлое: ленту публикаций с поиском и фильтрами,
честные числа, и пересобрать навигацию в четыре раздела вместо трёх неявных
экранов.

**Устройство.** Сервер получает два новых маршрута (`/history`, `/stats`) поверх
трёх новых функций базы. Клиент разбивается с одного файла в четыре обычных
скрипта (не модули — этого требует `node --check` в тесте) и получает нижнюю
навигацию с четырьмя разделами.

**Стек.** Python 3 + unittest, sqlite/Postgres через собственную обёртку;
клиент — обычный JS без сборки, CSS без препроцессора, инлайновый SVG.

## Общие ограничения

- CSP: `font-src 'self'`, `script-src 'self' https://telegram.org`,
  `connect-src 'self'`. Внешних шрифтов, скриптов и запросов нет.
- В `index.html` нет ни `<script>` без `src`, ни `<style>`. Атрибут `style="…"` можно.
- `app.js` остаётся, каждый файл `webapp/*.js` обязан проходить `node --check`.
- Модули ES запрещены: `node --check` разбирает поток как обычный скрипт.
- Палитра и скругления не меняются — это общий бренд с настольным 4xTube.
- Подстановка в SQL — только `?`; обёртка сама заменит на `%s`. Знак `%`
  для `LIKE` живёт в значении, а не в тексте запроса.
- Телеграмная публикация отличается от выложенной руками по `tg_message_id`:
  у ручной он пуст (`db.py:333-335`).
- Просмотры есть только у телеграма и только когда включён мак.
- Тесты запускаются: `python3 -m unittest discover -s tests -t . -v`

"""Лента вышедшего: страницы, поиск и фильтры.

База у владельца в Ирландии, и лента не может приезжать целиком. Значит
страницы и отбор считает сама база, а не клиент.
"""

import sqlite3
import sys
import unittest
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))


class _База:
    """Свежая база в памяти на каждый тест."""

    def __enter__(self):
        import db
        self.db = db
        self.связь = sqlite3.connect(":memory:")
        self.связь.row_factory = sqlite3.Row
        self.было = db.соединение
        db.соединение = self._дать
        for запрос in db.СХЕМА:
            self.связь.execute(запрос)
        for имя, тип in db.ДОБАВКИ:
            try:
                self.связь.execute(f"ALTER TABLE posts ADD COLUMN {имя} {тип}")
            except sqlite3.OperationalError:
                pass
        return self

    def _дать(self):
        внешний = self

        class Держатель:
            def __enter__(self_):
                return внешний.db._Связь(внешний.связь)

            def __exit__(self_, *_):
                внешний.связь.commit()
        return Держатель()

    def добавить(self, **поля):
        поля.setdefault("rubric", "разбор")
        поля.setdefault("title", "Заголовок")
        поля.setdefault("body", "Текст")
        поля.setdefault("status", "published")
        поля.setdefault("created_at", "2026-09-01T10:00:00")
        поля.setdefault("published_at", "2026-09-01T10:00:00")
        ключи = ", ".join(поля)
        места = ", ".join("?" for _ in поля)
        self.связь.execute(
            f"INSERT INTO posts ({ключи}) VALUES ({места})", tuple(поля.values()))
        self.связь.commit()

    def __exit__(self, *_):
        self.db.соединение = self.было
        self.связь.close()


class Лента(unittest.TestCase):

ПРЕДЕЛ_ЛЕНТЫ = 100          # больше одной страницей не отдаём никогда


def история(смещение: int = 0, сколько: int = 20, запрос: str | None = None,
            рубрика: str | None = None, площадка: str | None = None,
            дней: int | None = None,
            канал: str | None = None) -> tuple[list, int]:
    """Вышедшее постранично, с поиском и отбором.

    Считает база, а не клиент: она в Ирландии, и возить туда-сюда всю ленту
    ради двадцати строк на экране — это лишний поход через океан на каждую
    прокрутку.
    """
    from datetime import timedelta
    где = ["status='published'"]
    значения: list = []
    if запрос and запрос.strip():
        # Знак процента живёт в значении: в тексте запроса его нельзя,
        # там его съест замена «?» на «%s» для постгреса.
        где.append("LOWER(title) LIKE ?")
        значения.append(f"%{запрос.strip().lower()}%")
    if рубрика:
        где.append("rubric=?")
        значения.append(рубрика)
    if площадка == "телеграм":
        где.append("tg_message_id IS NOT NULL")
    elif площадка == "руками":
        где.append("tg_message_id IS NULL")
    if дней:
        порог = (datetime.now(timezone.utc) - timedelta(days=int(дней))
                 ).isoformat(timespec="seconds")
        где.append("published_at >= ?")
        значения.append(порог)
    if канал:
        # Пост, выложенный руками, канала не имеет вовсе: он ушёл в TikTok,
        # а channel у нас про телеграм. По строгому равенству такие посты
        # выпадали бы из ленты совсем.
        где.append("(channel=? OR channel IS NULL OR channel='')")
        значения.append(канал)
    условие = " AND ".join(где)
    сколько = max(1, min(int(сколько or 20), ПРЕДЕЛ_ЛЕНТЫ))
    смещение = max(0, int(смещение or 0))
    with соединение() as c:
        всего = c.execute(f"SELECT COUNT(*) AS н FROM posts WHERE {условие}",
                          tuple(значения)).fetchone()["н"]
        ряды = c.execute(
            f"SELECT * FROM posts WHERE {условие} ORDER BY id DESC"
            " LIMIT ? OFFSET ?", tuple(значения) + (сколько, смещение)).fetchall()
    return list(ряды), int(всего or 0)
```

def по_неделям(недель: int = 8, канал: str | None = None) -> list[dict]:
    """Штук и средний охват по неделям, от старой недели к свежей.

    Неделю считаем сами, а не запросом: `strftime` в sqlite и `date_trunc` в
    постгресе пишутся по-разному, и отчёт на маке разошёлся бы с облачным.
    Пустые недели остаются в ответе нулями — провал в выпуске это тоже
    сведения, и молчать о нём график не должен.
    """
    from datetime import timedelta
    недель = max(1, min(int(недель or 8), 52))
    сегодня = datetime.now(timezone.utc).date()
    # Начало текущей недели — понедельник.
    эта = сегодня - timedelta(days=сегодня.weekday())
    начала = [эта - timedelta(weeks=н) for н in range(недель - 1, -1, -1)]
    порог = datetime(начала[0].year, начала[0].month, начала[0].day,
                     tzinfo=timezone.utc).isoformat(timespec="seconds")
    условие = " AND (channel=? OR channel IS NULL OR channel='')" if канал else ""
    значения = (порог, канал) if канал else (порог,)
    with соединение() as c:
        ряды = c.execute(
            "SELECT published_at, views FROM posts WHERE status='published'"
            f" AND published_at >= ?{условие}", значения).fetchall()
    корзины = {н: {"неделя": н.isoformat(), "штук": 0, "сумма": 0, "с_числом": 0}
               for н in начала}
    for р in ряды:
        try:
            когда = datetime.fromisoformat(str(р["published_at"])).date()
        except (TypeError, ValueError):
            continue
        начало = когда - timedelta(days=когда.weekday())
        корзина = корзины.get(начало)
        if корзина is None:
            continue
        корзина["штук"] += 1
        просмотры = р["views"] if "views" in р.keys() else None
        if просмотры:
            корзина["сумма"] += int(просмотры)
            корзина["с_числом"] += 1
    итог = []
    for н in начала:
        к = корзины[н]
        итог.append({"неделя": к["неделя"], "штук": к["штук"],
                     "охват": round(к["сумма"] / к["с_числом"]) if к["с_числом"] else 0})
    return итог


def итоги(канал: str | None = None) -> dict:
    """Окна выпуска и охваты одним запросом.

    Одним, а не пятью: раздел «Вышло» открывается с телефона, и каждый
    лишний поход к базе за океан человек чувствует пальцами.
    """
    from datetime import timedelta
    сейчас = datetime.now(timezone.utc)
    неделя = (сейчас - timedelta(days=7)).isoformat(timespec="seconds")
    месяц = (сейчас - timedelta(days=30)).isoformat(timespec="seconds")
    прошлый = (сейчас - timedelta(days=60)).isoformat(timespec="seconds")
    свой = " AND (channel=? OR channel IS NULL OR channel='')" if канал else ""
    хвост = (канал,) if канал else ()
    with соединение() as c:
        ряд = c.execute(
            "SELECT COUNT(*) AS всего,"
            " COUNT(*) FILTER (WHERE published_at >= ?) AS за7,"
            " COUNT(*) FILTER (WHERE published_at >= ?) AS за30,"
            " AVG(views) FILTER (WHERE published_at >= ? AND views > 0)"
            "   AS охват30,"
            " AVG(views) FILTER (WHERE published_at >= ? AND published_at < ?"
            "   AND views > 0) AS охват_прошлые30,"
            " COUNT(*) FILTER (WHERE views > 0) AS с_числами"
            f" FROM posts WHERE status='published'{свой}",
            (неделя, месяц, месяц, прошлый, месяц) + хвост).fetchone()
    return {"всего": int(ряд["всего"] or 0), "за7": int(ряд["за7"] or 0),
            "за30": int(ряд["за30"] or 0),
            "охват30": round(float(ряд["охват30"] or 0)),
            "охват_прошлые30": round(float(ряд["охват_прошлые30"] or 0)),
            "с_числами": int(ряд["с_числами"] or 0)}
```

Разбор пути идёт лестницей `endswith` сверху вниз. Новый маршрут, чей

class _Подмена:
    def __init__(self, **что):
        self.что = что
        self.было = {}

    def __enter__(self):
        import db
        for имя, значение in self.что.items():
            self.было[имя] = getattr(db, имя)
            setattr(db, имя, значение)
        return self

    def __exit__(self, *_):
        import db
        for имя, значение in self.было.items():
            setattr(db, имя, значение)


def _ряд(ид: int, заголовок: str) -> dict:
