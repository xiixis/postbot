```python
# -*- coding: utf-8 -*-
"""Ключи Meta: получить, проверить, продлить.

Instagram и страница Facebook публикуются одним ключом страницы. Он бессрочный,
если выведен из долгоживущего ключа пользователя, поэтому получают его один раз:
владелец берёт короткий ключ в Graph API Explorer, кладёт его в .env строкой
META_USER_TOKEN, а этот скрипт меняет его на ключ страницы и находит номер
Instagram. Короткий ключ после обмена из .env убирается.

Threads живёт в своём приложении, и ключ у него на 60 дней. Продлевает его
рабочий раз в неделю (worker.py), а первый раз его получают так же: короткий
ключ строкой THREADS_SHORT_TOKEN, затем `threads-ключ`.

Ключи в терминал не печатаются, только их хвост.

    python3 tools_meta.py ключи            # короткий → ключ страницы, номер Instagram
    python3 tools_meta.py проверить        # чьи ключи лежат в .env
    python3 tools_meta.py threads-ключ     # короткий Threads → на 60 дней
    python3 tools_meta.py threads-продлить
"""

import sys
import time
from pathlib import Path

import requests
from dotenv import dotenv_values

ENV = Path(__file__).resolve().parent / ".env"
ГРАФ = "https://graph.facebook.com/v23.0"
THREADS = "https://graph.threads.net"
НЕДЕЛЯ = 7 * 86400


def скрыть(ключ: str) -> str:
    return f"…{ключ[-4:]}" if ключ else "(пусто)"


def записать_env(пары: dict[str, str], путь: Path = ENV) -> None:
    """Меняет строки ИМЯ=… на месте, новые дописывает в конец.

    Остальные строки, комментарии и порядок не трогаются: .env правят и руками.
    Пустое значение убирает строку совсем.
    """
    строки = путь.read_text(encoding="utf-8").splitlines() if путь.exists() else []
    осталось = dict(пары)
    итог = []
    for с in строки:
        имя = с.split("=", 1)[0].strip()
        if имя in осталось:
            значение = осталось.pop(имя)
            if значение:
                итог.append(f"{имя}={значение}")
            continue
        итог.append(с)
    итог += [f"{и}={з}" for и, з in осталось.items() if з]
    путь.write_text("\n".join(итог) + "\n", encoding="utf-8")


def _env(путь: Path = ENV) -> dict:
    return {к: (з or "").strip() for к, з in dotenv_values(путь).items()}


def _get(адрес: str, **поля) -> dict:
    r = requests.get(адрес, params=поля, timeout=60)
    тело = r.json() if r.content else {}
    if r.status_code >= 300 or "error" in тело:
        raise RuntimeError((тело.get("error") or {}).get("message") or r.text[:200])
    return тело


def ключи(путь: Path = ENV) -> None:
    """Короткий ключ пользователя → долгий → бессрочный ключ страницы."""
    e = _env(путь)
    for имя in ("META_APP_ID", "META_APP_SECRET", "META_USER_TOKEN"):
        if not e.get(имя):
            sys.exit(f"в .env нет {имя}")
    долгий = _get(f"{ГРАФ}/oauth/access_token", grant_type="fb_exchange_token",
                  client_id=e["META_APP_ID"], client_secret=e["META_APP_SECRET"],
                  fb_exchange_token=e["META_USER_TOKEN"])["access_token"]
    страницы = _get(f"{ГРАФ}/me/accounts", access_token=долгий,
                    fields="id,name,access_token,instagram_business_account{id,username}"
                    ).get("data") or []
    if not страницы:
        sys.exit("за ключом нет ни одной страницы: при входе отметьте страницу канала")
    for н, с in enumerate(страницы):
        иг = (с.get("instagram_business_account") or {}).get("username", "—")
        print(f"{н}: {с['name']} · Instagram @{иг}")
    выбор = 0 if len(страницы) == 1 else int(input("номер страницы: "))
    с = страницы[выбор]
    иг = (с.get("instagram_business_account") or {}).get("id", "")
    if not иг:
        sys.exit("к странице не привязан профессиональный Instagram")
    записать_env({"META_PAGE_ID": с["id"], "META_PAGE_TOKEN": с["access_token"],
                  "IG_USER_ID": иг, "META_USER_TOKEN": ""}, путь)
    print(f"записано: страница {с['name']}, ключ {скрыть(с['access_token'])}, "
          f"Instagram {иг}; короткий ключ убран")


def проверить(путь: Path = ENV) -> None:
    e = _env(путь)
    if e.get("META_PAGE_TOKEN"):
        о = _get(f"{ГРАФ}/debug_token", input_token=e["META_PAGE_TOKEN"],
                 access_token=e["META_PAGE_TOKEN"])["data"]
        срок = "бессрочный" if not о.get("expires_at") else time.strftime(
            "%d.%m.%Y", time.localtime(о["expires_at"]))
        print(f"страница: годен={о.get('is_valid')} · {срок} · права {о.get('scopes')}")
        иг = _get(f"{ГРАФ}/{e['IG_USER_ID']}", fields="username",
                  access_token=e["META_PAGE_TOKEN"])
        print(f"Instagram: @{иг.get('username')}")
    if e.get("THREADS_TOKEN"):
        т = _get(f"{THREADS}/v1.0/me", fields="id,username",
                 access_token=e["THREADS_TOKEN"])
        print(f"Threads: @{т.get('username')} · номер {т.get('id')}")


def threads_ключ(путь: Path = ENV) -> None:
    """Короткий ключ Threads → на 60 дней, и номер пользователя."""
    e = _env(путь)
    for имя in ("THREADS_APP_SECRET", "THREADS_SHORT_TOKEN"):
        if not e.get(имя):
            sys.exit(f"в .env нет {имя}")
    долгий = _get(f"{THREADS}/access_token", grant_type="th_exchange_token",
                  client_secret=e["THREADS_APP_SECRET"],
                  access_token=e["THREADS_SHORT_TOKEN"])["access_token"]
    кто = _get(f"{THREADS}/v1.0/me", fields="id,username", access_token=долгий)
    записать_env({"THREADS_TOKEN": долгий, "THREADS_USER_ID": str(кто["id"]),
                  "THREADS_TOKEN_UPDATED": str(int(time.time())),
                  "THREADS_SHORT_TOKEN": ""}, путь)
    print(f"Threads @{кто.get('username')}: ключ {скрыть(долгий)} на 60 дней")


def пора_продлить(сейчас: float, путь: Path = ENV) -> bool:
    e = _env(путь) if путь.exists() else {}
    if not e.get("THREADS_TOKEN"):
        return False
    было = float(e.get("THREADS_TOKEN_UPDATED") or 0)
    return сейчас - было >= НЕДЕЛЯ


def продлить_threads(путь: Path = ENV) -> bool:
    """Продлевает ключ Threads ещё на 60 дней. Работает только с ключом старше суток."""
    e = _env(путь)
    новый = _get(f"{THREADS}/refresh_access_token", grant_type="th_refresh_token",
                 access_token=e["THREADS_TOKEN"])["access_token"]
    записать_env({"THREADS_TOKEN": новый,
                  "THREADS_TOKEN_UPDATED": str(int(time.time()))}, путь)
    import os
    os.environ["THREADS_TOKEN"] = новый
    try:
        import threads
        threads.ТОКЕН = новый
    except ImportError:
        pass
    return True


if __name__ == "__main__":
    что = sys.argv[1] if len(sys.argv) > 1 else ""
    {"ключи": ключи, "проверить": проверить, "threads-ключ": threads_ключ,
     "threads-продлить": lambda: print("продлён") if продлить_threads() else None
     }.get(что, lambda: sys.exit(__doc__))()
```

```python
# -*- coding: utf-8 -*-
"""Instagram: порядок запросов карусели, ожидание обработки, отказы."""

import sys
import unittest
from pathlib import Path
from unittest import mock

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

import instagram                                             # noqa: E402


class Ответ:
    def __init__(self, тело, код=200):
        self._т, self.status_code, self.content, self.text = тело, код, b"1", str(тело)

    def json(self):
        return self._т


class Сеть:
    """Запоминает запросы и отвечает по порядку."""

    def __init__(self, статусы=("FINISHED",)):
        self.post_вызовы, self.статусы, self.номер = [], list(статусы), 0

    def post(self, адрес, data=None, timeout=None):
        self.post_вызовы.append((адрес.rsplit("/", 1)[-1], dict(data)))
        self.номер += 1
        return Ответ({"id": f"c{self.номер}"})

    def get(self, адрес, params=None, timeout=None):
        if params.get("fields") == "permalink":
            return Ответ({"permalink": "https://www.instagram.com/p/XYZ/"})
        return Ответ({"status_code": self.статусы.pop(0) if self.статусы else "FINISHED"})


class Публикация(unittest.TestCase):
    def setUp(self):
        self.сеть = Сеть()
        for имя, знач in (("ТОКЕН", "t"), ("ПОЛЬЗОВАТЕЛЬ", "17")):
            p = mock.patch.object(instagram, имя, знач)
            p.start()
            self.addCleanup(p.stop)
        for имя in ("post", "get"):
            p = mock.patch.object(instagram.requests, имя, getattr(self.сеть, имя))
            p.start()
            self.addCleanup(p.stop)
        p = mock.patch.object(instagram.time, "sleep", lambda *_: None)
        p.start()
        self.addCleanup(p.stop)

    def test_карусель(self):
        след = instagram.опубликовать("подпись #тег", ["https://a/1.jpg", "https://a/2.jpg"])
        пути = [в[0] for в in self.сеть.post_вызовы]
        self.assertEqual(пути, ["media", "media", "media", "media_publish"])
        self.assertEqual(self.сеть.post_вызовы[0][1]["is_carousel_item"], "true")
        сборка = self.сеть.post_вызовы[2][1]
        self.assertEqual(сборка["media_type"], "CAROUSEL")
        self.assertEqual(сборка["children"], "c1,c2")
        self.assertEqual(сборка["caption"], "подпись #тег")
        self.assertEqual(self.сеть.post_вызовы[3][1]["creation_id"], "c3")
        self.assertEqual(след, "https://www.instagram.com/p/XYZ/")

    def test_один_кадр_без_карусели(self):
        instagram.опубликовать("п", ["https://a/1.jpg"])
        пути = [в[0] for в in self.сеть.post_вызовы]
        self.assertEqual(пути, ["media", "media_publish"])
        self.assertNotIn("is_carousel_item", self.сеть.post_вызовы[0][1])
        self.assertEqual(self.сеть.post_вызовы[0][1]["caption"], "п")

    def test_ждёт_обработки(self):
        self.сеть.статусы = ["IN_PROGRESS", "IN_PROGRESS", "FINISHED"]
        instagram.опубликовать("п", ["https://a/1.jpg"])
        self.assertEqual(self.сеть.post_вызовы[-1][0], "media_publish")

    def test_ошибка_обработки(self):
        self.сеть.статусы = ["ERROR"]
        with self.assertRaises(instagram.ОшибкаInstagram):
            instagram.опубликовать("п", ["https://a/1.jpg"])

    def test_одиннадцать_кадров_отказ(self):
        with self.assertRaisesRegex(instagram.ОшибкаInstagram, "10"):
            instagram.опубликовать("п", [f"https://a/{н}.jpg" for н in range(11)])
        self.assertEqual(self.сеть.post_вызовы, [])

    def test_без_картинок_отказ(self):
        with self.assertRaises(instagram.ОшибкаInstagram):
            instagram.опубликовать("п", [])

    def test_длинная_подпись_режется(self):
        instagram.опубликовать("я" * 3000, ["https://a/1.jpg"])
        self.assertEqual(len(self.сеть.post_вызовы[0][1]["caption"]), instagram.ПРЕДЕЛ)

    def test_ошибка_graph_api(self):
        def плохо(*_а, **_к):
            return Ответ({"error": {"message": "Invalid aspect ratio"}}, 400)
        with mock.patch.object(instagram.requests, "post", плохо):
            with self.assertRaisesRegex(instagram.ОшибкаInstagram, "aspect"):
                instagram.опубликовать("п", ["https://a/1.jpg"])


if __name__ == "__main__":
    unittest.main()
```

```python
# -*- coding: utf-8 -*-
"""Публикация в Instagram через Graph API.

Устроено как Threads: каждый кадр сперва становится заготовкой, заготовки
собираются в карусель, и только потом она публикуется. Картинки Instagram
забирает сам по публичной ссылке, файлы он не принимает.

Требования площадки: только JPEG, не больше десяти кадров, пропорции от 4:5
до 1.91:1. Наш старый 3:4 он отвергает, поэтому под Instagram кадры рисуются
отдельно (площадки.py).

Ключ здесь ключ страницы, а не Instagram: аккаунт привязан к странице Facebook,
и один ключ страницы открывает оба. Получает его tools_meta.py.
"""

import os
import time

import requests

БАЗА = "https://graph.facebook.com/v23.0"
ТОКЕН = os.getenv("META_PAGE_TOKEN", "").strip()
ПОЛЬЗОВАТЕЛЬ = os.getenv("IG_USER_ID", "").strip()

КАДРОВ = 10
ПРЕДЕЛ = 2200
ЖДАТЬ = 300           # секунд на обработку заготовки, потом сдаёмся


class ОшибкаInstagram(RuntimeError):
    pass


def настроен() -> bool:
    return bool(ТОКЕН and ПОЛЬЗОВАТЕЛЬ)


def _разобрать(r) -> dict:
    try:
        тело = r.json()
    except ValueError:
        тело = {"error": {"message": r.text[:300]}}
    if r.status_code >= 300 or "error" in тело:
        беда = (тело.get("error") or {}).get("message") or r.text[:200]
        raise ОшибкаInstagram(f"{r.status_code}: {беда}")
    return тело


def _post(путь: str, поля: dict) -> dict:
    return _разобрать(requests.post(f"{БАЗА}/{путь}",
                                    data={**поля, "access_token": ТОКЕН}, timeout=120))


def _get(путь: str, поля: str) -> dict:
    return _разобрать(requests.get(f"{БАЗА}/{путь}",
                                   params={"fields": поля, "access_token": ТОКЕН},
                                   timeout=60))


def _заготовка(поля: dict) -> str:
    номер = _post(f"{ПОЛЬЗОВАТЕЛЬ}/media", поля).get("id")
    if not номер:
        raise ОшибкаInstagram("Instagram не вернул номер заготовки")
    return str(номер)


def _дождаться(номер: str) -> None:
    """Instagram скачивает картинки сам, и публиковать до конца обработки нельзя."""
    до = time.time() + ЖДАТЬ
    while True:
        статус = _get(номер, "status_code").get("status_code")
        if статус == "FINISHED":
            return
        if статус in ("ERROR", "EXPIRED"):
            raise ОшибкаInstagram(f"Instagram не принял кадры: {статус}")
        if time.time() > до:
            raise ОшибкаInstagram("Instagram не обработал кадры за пять минут")
        time.sleep(3)


def опубликовать(подпись: str, картинки: list | None = None) -> str:
    """Публикует пост. Возвращает адрес поста, а если его не отдали, номер."""
    if not настроен():
        raise ОшибкаInstagram("не заданы META_PAGE_TOKEN и IG_USER_ID")
    ссылки = [к for к in (картинки or []) if isinstance(к, str) and к.startswith("http")]
    if not ссылки:
        raise ОшибкаInstagram("в Instagram нечего публиковать: нет кадров по ссылке")
    if len(ссылки) > КАДРОВ:
        # Резать молча нельзя: пропадёт конец поста, а с ним вывод и призыв.
        raise ОшибкаInstagram(f"в карусели {len(ссылки)} кадров, Instagram берёт до {КАДРОВ}")
    подпись = (подпись or "").strip()
    if len(подпись) > ПРЕДЕЛ:
        подпись = подпись[:ПРЕДЕЛ - 1].rstrip() + "…"

    if len(ссылки) > 1:
        дети = [_заготовка({"image_url": с, "is_carousel_item": "true"}) for с in ссылки]
        заготовка = _заготовка({"media_type": "CAROUSEL",
                                "children": ",".join(дети), "caption": подпись})
    else:
        заготовка = _заготовка({"image_url": ссылки[0], "caption": подпись})
    _дождаться(заготовка)

    номер = str(_post(f"{ПОЛЬЗОВАТЕЛЬ}/media_publish",
                      {"creation_id": заготовка}).get("id", ""))
    try:
        return _get(номер, "permalink").get("permalink") or номер
    except ОшибкаInstagram:
        return номер          # пост вышел, адрес не главное
```

```python
# -*- coding: utf-8 -*-
"""Публикация на страницу Facebook.

Пост с несколькими кадрами собирается в два хода: каждый кадр загружается на
страницу неопубликованным, а потом выходит одна запись, к которой они
приложены. Если опубликовать кадры сразу, на странице окажется десять
отдельных постов.

Ключ тот же, что у Instagram: ключ страницы из tools_meta.py.
"""

import json
import os

import requests

БАЗА = "https://graph.facebook.com/v23.0"
ТОКЕН = os.getenv("META_PAGE_TOKEN", "").strip()
СТРАНИЦА = os.getenv("META_PAGE_ID", "").strip()


class ОшибкаFacebook(RuntimeError):
    pass


def настроен() -> bool:
    return bool(ТОКЕН and СТРАНИЦА)


def _post(путь: str, поля: dict) -> dict:
    r = requests.post(f"{БАЗА}/{путь}", data={**поля, "access_token": ТОКЕН}, timeout=120)
    try:
        тело = r.json()
    except ValueError:
        тело = {"error": {"message": r.text[:300]}}
    if r.status_code >= 300 or "error" in тело:
        беда = (тело.get("error") or {}).get("message") or r.text[:200]
        raise ОшибкаFacebook(f"{r.status_code}: {беда}")
    return тело


def опубликовать(текст: str, картинки: list | None = None) -> str:
    """Публикует пост на странице. Возвращает его номер."""
    if not настроен():
        raise ОшибкаFacebook("не заданы META_PAGE_TOKEN и META_PAGE_ID")
    текст = (текст or "").strip()
    ссылки = [к for к in (картинки or []) if isinstance(к, str) and к.startswith("http")]
    if not текст and not ссылки:
        raise ОшибкаFacebook("пустой пост")
    поля = {"message": текст}
    for н, с in enumerate(ссылки):
        кадр = _post(f"{СТРАНИЦА}/photos", {"url": с, "published": "false"})["id"]
        поля[f"attached_media[{н}]"] = json.dumps({"media_fbid": кадр})
    return str(_post(f"{СТРАНИЦА}/feed", поля).get("id", ""))
```
